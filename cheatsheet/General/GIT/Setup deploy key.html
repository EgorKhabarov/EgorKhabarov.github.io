ed25519

bash
git config --global user.name "github-username"
git config --global user.email "github-email"
ssh-keygen -t ed25519 -C "github-email"


eval "$(ssh-agent -s)"
ssh-add ~/.ssh/id_ed25519

rsa

bash
git config --global user.name "github-username"
git config --global user.email "github-email"
ssh-keygen -t rsa -b 4096 -C "github-email"


eval "$(ssh-agent -s)"
ssh-add ~/.ssh/id_rsa
